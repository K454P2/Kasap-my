function lihatRincianPesanan(p){
  var subtotal = Number(p.total||0);
  var ongkir = Number(p.ongkir||0);
  var platform = Math.round(subtotal * 0.05);
  var layanan = 3600;
  var pajak = 400;
  var akhir = subtotal + ongkir - platform - layanan - pajak;
  if(akhir < 0) akhir = subtotal;
  document.getElementById("rNama").textContent = p.nama_pembeli || p.nama || "Pembeli";
  document.getElementById("rHP").textContent = p.no_hp || "08xxxxxx05";
  document.getElementById("rAlamat").textContent = p.alamat || "Alamat belum diisi pembeli";
  document.getElementById("rResi").textContent = "Pos Reguler: " + (p.resi || "- Belum ada resi -");
  document.getElementById("rStatus").textContent = p.status==="Selesai" ? "Pesanan tiba di alamat tujuan." : p.status==="Dikirim" ? "Pesanan sedang dikirim." : "Pesanan Baru - " + (p.nama_produk||"");
  document.getElementById("rTanggal").textContent = p.created_at ? new Date(p.created_at).toLocaleString("id-ID") : "-";
  document.getElementById("rSubtotal").textContent = "Rp " + subtotal.toLocaleString("id-ID");
  document.getElementById("rOngkir").textContent = "Rp " + ongkir.toLocaleString("id-ID");
  document.getElementById("rPlatform").textContent = "-Rp " + platform.toLocaleString("id-ID");
  document.getElementById("rLayanan").textContent = "-Rp " + layanan.toLocaleString("id-ID");
  document.getElementById("rPajak").textContent = "-Rp " + pajak.toLocaleString("id-ID");
  document.getElementById("rHasil").textContent = "Rp " + akhir.toLocaleString("id-ID");
  document.getElementById("rSesuai").textContent = "Rp 0";
  document.getElementById("rAkhir").textContent = "Rp " + akhir.toLocaleString("id-ID");
  document.getElementById("rBayar").textContent = "Rp " + (subtotal+ongkir).toLocaleString("id-ID");
  var box = document.getElementById("rincianBox");
  box.style.display = "block";
  box.scrollIntoView();
}
</script>
